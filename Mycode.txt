int main(void) {
    PatientList patients;
    DoctorList doctors;
    AppointmentList appointments;
    WardList wards;
    BillList bills;
    int choice = 0;

    init_patient_list(&patients);
    init_doctor_list(&doctors);
    init_appointment_list(&appointments);
    init_ward_list(&wards);
    init_bill_list(&bills);

    load_all_data(&patients, &doctors, &appointments, &wards, &bills);

    if (patients.count == 0 && doctors.count == 0 && wards.count == 0) {
        seed_default_data(&patients, &doctors, &appointments, &wards, &bills);
        save_all_data(&patients, &doctors, &appointments, &wards, &bills);
        printf("[Info] Initialized default hospital records.\n");
    }

    do {
        print_header("HOSPITAL MANAGEMENT SYSTEM");
        printf("  1. Patient Management\n");
        printf("  2. Doctor Management\n");
        printf("  3. Appointment Scheduling\n");
        printf("  4. Ward & Bed Management\n");
        printf("  5. Billing & Payments\n");
        printf("  6. Reports & Analytics\n");
        printf("  7. Save All Records Now\n");
        printf("  0. Save & Exit\n");
        print_separator('-', 52);

        if (!read_int_range("Select an option: ", 0, 7, &choice)) {
            printf("[Error] Invalid input. Enter a number from 0 to 7.\n");
            continue;
        }

        switch (choice) {
            case 1:
                patient_menu(&patients, &doctors, &wards);
                break;
            case 2:
                doctor_menu(&doctors, &appointments);
                break;
            case 3:
                appointment_menu(&appointments, &patients, &doctors);
                break;
            case 4:
                ward_menu(&wards, &patients);
                break;
            case 5:
                billing_menu(&bills, &patients, &doctors, &wards);
                break;
            case 6:
                reports_menu(&patients, &doctors, &appointments, &wards, &bills);
                break;
            case 7:
                save_all_data(&patients, &doctors, &appointments, &wards, &bills);
                printf("[Success] All records saved to disk.\n");
                pause_for_enter();
                break;
            case 0:
                save_all_data(&patients, &doctors, &appointments, &wards, &bills);
                printf("[Success] Data saved. Exiting system.\n");
                break;
            default:
                break;
        }
    } while (choice != 0);

    free_patient_list(&patients);
    free_doctor_list(&doctors);
    free_appointment_list(&appointments);
    free_ward_list(&wards);
    free_bill_list(&bills);

    return 0;
}

static void clear_stdin(void) {
    int ch;
    while ((ch = getchar()) != '\n' && ch != EOF) {
        /* discard remaining characters */
    }
}

static void trim_newline(char *str) {
    size_t len;
    if (!str) return;
    len = strlen(str);
    while (len > 0 && (str[len - 1] == '\n' || str[len - 1] == '\r')) {
        str[--len] = '\0';
    }
}

static void trim_whitespace(char *str) {
    size_t start = 0;
    size_t len;
    if (!str) return;
    len = strlen(str);
    while (start < len && isspace((unsigned char)str[start])) {
        start++;
    }
    if (start > 0) {
        memmove(str, str + start, len - start + 1);
        len -= start;
    }
    while (len > 0 && isspace((unsigned char)str[len - 1])) {
        str[--len] = '\0';
    }
}

static int read_line(const char *prompt, char *buffer, size_t size) {
    if (!buffer || size == 0) return 0;
    if (prompt) {
        printf("%s", prompt);
        fflush(stdout);
    }
    if (!fgets(buffer, (int)size, stdin)) {
        clearerr(stdin);
        buffer[0] = '\0';
        return 0;
    }
    if (!strchr(buffer, '\n')) {
        clear_stdin();
    }
    trim_newline(buffer);
    trim_whitespace(buffer);
    return 1;
}

static int read_int_range(const char *prompt, int min_val, int max_val, int *out_val) {
    char line[64];
    char *endptr = NULL;
    long val;

    if (!read_line(prompt, line, sizeof(line)) || line[0] == '\0') {
        return 0;
    }
    val = strtol(line, &endptr, 10);
    if (endptr == line || *endptr != '\0') {
        return 0;
    }
    if (val < (long)min_val || val > (long)max_val) {
        return 0;
    }
    *out_val = (int)val;
    return 1;
}

static int read_double_min(const char *prompt, double min_val, double *out_val) {
    char line[64];
    char *endptr = NULL;
    double val;

    if (!read_line(prompt, line, sizeof(line)) || line[0] == '\0') {
        return 0;
    }
    val = strtod(line, &endptr);
    if (endptr == line || *endptr != '\0') {
        return 0;
    }
    if (val < min_val) {
        return 0;
    }
    *out_val = val;
    return 1;
}

static int read_char_choice(const char *prompt, const char *valid_chars, char *out_ch) {
    char line[16];
    char ch;
    if (!read_line(prompt, line, sizeof(line)) || line[0] == '\0') {
        return 0;
    }
    ch = (char)toupper((unsigned char)line[0]);
    if (strchr(valid_chars, ch) == NULL) {
        return 0;
    }
    *out_ch = ch;
    return 1;
}

static int is_leap_year(int year) {
    return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
}

static int is_valid_date(const char *date_str) {
    int year, month, day;
    int days_in_month[] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

    if (!date_str || strlen(date_str) != 10) return 0;
    if (date_str[4] != '-' || date_str[7] != '-') return 0;
    for (int i = 0; i < 10; i++) {
        if (i == 4 || i == 7) continue;
        if (!isdigit((unsigned char)date_str[i])) return 0;
    }
    if (sscanf(date_str, "%4d-%2d-%2d", &year, &month, &day) != 3) return 0;
    if (year < 1900 || year > 2100) return 0;
    if (month < 1 || month > 12) return 0;
    if (month == 2 && is_leap_year(year)) {
        days_in_month[2] = 29;
    }
    if (day < 1 || day > days_in_month[month]) return 0;
    return 1;
}

static void get_current_date(char *buffer, size_t size) {
    time_t now = time(NULL);
    struct tm *tm_info = localtime(&now);
    if (!buffer || size == 0) return;
    if (tm_info) {
        strftime(buffer, size, "%Y-%m-%d", tm_info);
    } else {
        snprintf(buffer, size, "2026-01-01");
    }
}

static void to_lower_copy(const char *src, char *dst, size_t dst_size) {
    size_t i = 0;
    if (!dst || dst_size == 0) return;
    if (!src) {
        dst[0] = '\0';
        return;
    }
    while (src[i] != '\0' && i + 1 < dst_size) {
        dst[i] = (char)tolower((unsigned char)src[i]);
        i++;
    }
    dst[i] = '\0';
}

static int contains_case_insensitive(const char *haystack, const char *needle) {
    char h_low[256];
    char n_low[256];
    if (!haystack || !needle) return 0;
    if (needle[0] == '\0') return 1;
    to_lower_copy(haystack, h_low, sizeof(h_low));
    to_lower_copy(needle, n_low, sizeof(n_low));
    return strstr(h_low, n_low) != NULL;
}

static void print_separator(char ch, int length) {
    for (int i = 0; i < length; i++) {
        putchar(ch);
    }
    putchar('\n');
}

static void print_header(const char *title) {
    putchar('\n');
    print_separator('=', 52);
    printf("  %s\n", title);
    print_separator('=', 52);
}

static void pause_for_enter(void) {
    char buf[8];
    read_line("\nPress Enter to continue...", buf, sizeof(buf));
}

static int save_patients(const PatientList *list, const char *filename) {
    FILE *fp = fopen(filename, "wb");
    if (!fp) return 0;
    if (fwrite(&list->next_id, sizeof(int), 1, fp) != 1 ||
        fwrite(&list->count, sizeof(size_t), 1, fp) != 1) {
        fclose(fp);
        return 0;
    }
    if (list->count > 0 && fwrite(list->items, sizeof(Patient), list->count, fp) != list->count) {
        fclose(fp);
        return 0;
    }
    fclose(fp);
    return 1;
}

static int load_patients(PatientList *list, const char *filename) {
    FILE *fp = fopen(filename, "rb");
    size_t file_count = 0;
    int next_id = 1001;

    if (!fp) return 0;
    if (fread(&next_id, sizeof(int), 1, fp) != 1 ||
        fread(&file_count, sizeof(size_t), 1, fp) != 1) {
        fclose(fp);
        return 0;
    }
    free_patient_list(list);
    init_patient_list(list);
    list->next_id = next_id;

    for (size_t i = 0; i < file_count; i++) {
        Patient p;
        if (fread(&p, sizeof(Patient), 1, fp) != 1) break;
        if (ensure_patient_capacity(list)) {
            list->items[list->count++] = p;
        }
    }
    fclose(fp);
    return 1;
}

static void patient_menu(PatientList *pl, DoctorList *dl, WardList *wl) {
    int choice = 0;
    do {
        print_header("PATIENT MANAGEMENT");
        printf("  1. Register / Admit Patient\n");
        printf("  2. View All Patients\n");
        printf("  3. Search Patients (ID, Name, Diagnosis)\n");
        printf("  4. Update Patient Record\n");
        printf("  5. Discharge Patient\n");
        printf("  6. Sort Patients by Name\n");
        printf("  7. Sort Patients by ID\n");
        printf("  0. Back to Main Menu\n");
        print_separator('-', 52);

        if (!read_int_range("Select option: ", 0, 7, &choice)) {
            printf("[Error] Enter a valid option (0-7).\n");
            continue;
        }

        switch (choice) {
            case 1: admit_patient(pl, dl, wl); pause_for_enter(); break;
            case 2: list_patients(pl, dl, wl); pause_for_enter(); break;
            case 3: search_patients(pl, dl); pause_for_enter(); break;
            case 4: update_patient_record(pl, dl, wl); pause_for_enter(); break;
            case 5: discharge_patient(pl, wl); pause_for_enter(); break;
            case 6: sort_patients_by_name(pl); printf("[Info] Patients sorted by name.\n"); pause_for_enter(); break;
            case 7: sort_patients_by_id(pl); printf("[Info] Patients sorted by ID.\n"); pause_for_enter(); break;
            case 0: break;
            default: break;
        }
    } while (choice != 0);
}

static void admit_patient(PatientList *pl, DoctorList *dl, WardList *wl) {
    Patient p;
    int doc_id = 0;
    int status_choice = 1;

    memset(&p, 0, sizeof(Patient));
    p.id = pl->next_id;
    p.active = 1;
    p.assigned_ward_id = -1;

    print_header("REGISTER / ADMIT PATIENT");
    if (!read_line("Full Name: ", p.full_name, sizeof(p.full_name)) || p.full_name[0] == '\0') {
        printf("[Error] Patient name cannot be empty.\n");
        return;
    }
    if (!read_int_range("Age (0-130): ", 0, 130, &p.age)) {
        printf("[Error] Invalid age.\n");
        return;
    }
    if (!read_char_choice("Gender (M/F/O): ", "MFO", &p.gender)) {
        printf("[Error] Invalid gender.\n");
        return;
    }
    read_line("Blood Group (e.g., O+, A-, B+): ", p.blood_group, sizeof(p.blood_group));
    read_line("Phone Number: ", p.phone, sizeof(p.phone));
    read_line("Address: ", p.address, sizeof(p.address));
    read_line("Primary Diagnosis / Complaint: ", p.diagnosis, sizeof(p.diagnosis));

    read_line("Admission Date (YYYY-MM-DD, blank for today): ", p.admission_date, sizeof(p.admission_date));
    if (p.admission_date[0] == '\0') {
        get_current_date(p.admission_date, sizeof(p.admission_date));
    } else if (!is_valid_date(p.admission_date)) {
        printf("[Error] Invalid date format. Use YYYY-MM-DD.\n");
        return;
    }

    if (dl->count > 0) {
        printf("\nAvailable Doctors:\n");
        for (size_t i = 0; i < dl->count; i++) {
            if (dl->items[i].active) {
                printf("  ID %d: %s (%s) - %s\n",
                       dl->items[i].id,
                       dl->items[i].full_name,
                       dl->items[i].specialization,
                       dl->items[i].available ? "Available" : "Unavailable");
            }
        }
        if (read_int_range("Assign Doctor ID (0 for none): ", 0, 999999, &doc_id)) {
            if (doc_id != 0 && find_doctor_by_id(dl, doc_id) == NULL) {
                printf("[Warning] Doctor ID %d not found; setting unassigned.\n", doc_id);
                doc_id = 0;
            }
        }
    }
    p.assigned_doctor_id = doc_id;

    printf("\nAdmission Type:\n  1. Outpatient\n  2. Inpatient\n");
    if (!read_int_range("Select type (1-2): ", 1, 2, &status_choice)) {
        status_choice = 1;
    }
    p.status = (status_choice == 2) ? PATIENT_INPATIENT : PATIENT_OUTPATIENT;

    if (p.status == PATIENT_INPATIENT && wl->count > 0) {
        int ward_num = 0;
        printf("\nAvailable Wards:\n");
        for (size_t i = 0; i < wl->count; i++) {
            Ward *w = &wl->items[i];
            if (w->active) {
                printf("  Ward %d: %-22s (%d/%d beds occupied, $%.2f/day)\n",
                       w->ward_id, w->ward_name, w->occupied_beds, w->total_beds, w->daily_rate);
            }
        }
        if (read_int_range("Assign Ward ID (0 to skip): ", 0, 999999, &ward_num) && ward_num != 0) {
            Ward *selected_ward = NULL;
            for (size_t i = 0; i < wl->count; i++) {
                if (wl->items[i].active && wl->items[i].ward_id == ward_num) {
                    selected_ward = &wl->items[i];
                    break;
                }
            }
            if (selected_ward && selected_ward->occupied_beds < selected_ward->total_beds) {
                selected_ward->occupied_beds++;
                p.assigned_ward_id = selected_ward->ward_id;
                printf("[Info] Assigned Bed in Ward %d (%s).\n",
                       selected_ward->ward_id, selected_ward->ward_name);
            } else {
                printf("[Warning] Ward unavailable or full; patient admitted without ward bed.\n");
            }
        }
    }

    if (!ensure_patient_capacity(pl)) {
        printf("[Error] Memory allocation failed.\n");
        return;
    }
    pl->items[pl->count++] = p;
    pl->next_id++;
    printf("[Success] Patient registered with ID %d.\n", p.id);
}

static void print_patient_row(const Patient *p, const DoctorList *dl, const WardList *wl) {
    const Doctor *doc = (p->assigned_doctor_id > 0) ? find_doctor_by_id((DoctorList *)dl, p->assigned_doctor_id) : NULL;
    const Ward *ward = NULL;
    if (p->assigned_ward_id > 0) {
        for (size_t i = 0; i < wl->count; i++) {
            if (wl->items[i].active && wl->items[i].ward_id == p->assigned_ward_id) {
                ward = &wl->items[i];
                break;
            }
        }
    }

    printf("ID: %-5d | Name: %-22s | Age/Sex: %2d/%c | Status: %-10s\n",
           p->id, p->full_name, p->age, p->gender, patient_status_str(p->status));
    printf("   Blood: %-4s | Phone: %-16s | Admitted: %s | Ward: %s\n",
           p->blood_group[0] ? p->blood_group : "N/A",
           p->phone[0] ? p->phone : "N/A",
           p->admission_date,
           ward ? ward->ward_name : "None");
    printf("   Doctor: %-22s | Diagnosis: %s\n",
           doc ? doc->full_name : "Unassigned",
           p->diagnosis[0] ? p->diagnosis : "None recorded");
}

static void list_patients(const PatientList *pl, const DoctorList *dl, const WardList *wl) {
    size_t shown = 0;
    print_header("PATIENT DIRECTORY");
    for (size_t i = 0; i < pl->count; i++) {
        if (pl->items[i].active) {
            print_patient_row(&pl->items[i], dl, wl);
            print_separator('-', 52);
            shown++;
        }
    }
    if (shown == 0) {
        printf("No patient records found.\n");
    } else {
        printf("Total active patient records: %zu\n", shown);
    }
}

static void search_patients(const PatientList *pl, const DoctorList *dl) {
    char query[128];
    int search_id = 0;
    size_t matches = 0;

    print_header("SEARCH PATIENTS");
    if (!read_line("Enter ID, Name, or Diagnosis keyword: ", query, sizeof(query)) || query[0] == '\0') {
        printf("[Error] Search query cannot be empty.\n");
        return;
    }

    search_id = atoi(query);
    for (size_t i = 0; i < pl->count; i++) {
        const Patient *p = &pl->items[i];
        if (!p->active) continue;
        if ((search_id > 0 && p->id == search_id) ||
            contains_case_insensitive(p->full_name, query) ||
            contains_case_insensitive(p->diagnosis, query)) {
            const Doctor *doc = (p->assigned_doctor_id > 0) ? find_doctor_by_id((DoctorList *)dl, p->assigned_doctor_id) : NULL;
            printf("  [%d] %-22s | %-10s | Doc: %-18s | Dx: %s\n",
                   p->id, p->full_name, patient_status_str(p->status),
                   doc ? doc->full_name : "None", p->diagnosis);
            matches++;
        }
    }
    printf("Matches found: %zu\n", matches);
}

static void update_patient_record(PatientList *pl, DoctorList *dl, WardList *wl) {
    int id = 0;
    Patient *p;
    char buf[MAX_DIAG_LEN];

    print_header("UPDATE PATIENT RECORD");
    if (!read_int_range("Enter Patient ID: ", 1, 999999, &id)) {
        printf("[Error] Invalid ID.\n");
        return;
    }
    p = find_patient_by_id(pl, id);
    if (!p) {
        printf("[Error] Patient ID %d not found.\n", id);
        return;
    }

    printf("Updating %s (ID %d). Leave blank to keep current value.\n", p->full_name, p->id);
    if (read_line("New Phone: ", buf, sizeof(buf)) && buf[0] != '\0') {
        snprintf(p->phone, sizeof(p->phone), "%s", buf);
    }
    if (read_line("New Address: ", buf, sizeof(buf)) && buf[0] != '\0') {
        snprintf(p->address, sizeof(p->address), "%s", buf);
    }
    if (read_line("New Diagnosis / Notes: ", buf, sizeof(buf)) && buf[0] != '\0') {
        snprintf(p->diagnosis, sizeof(p->diagnosis), "%s", buf);
    }

    if (dl->count > 0) {
        int new_doc = 0;
        if (read_int_range("New Assigned Doctor ID (0 to keep current): ", 0, 999999, &new_doc) && new_doc != 0) {
            if (find_doctor_by_id(dl, new_doc) != NULL) {
                p->assigned_doctor_id = new_doc;
            } else {
                printf("[Warning] Doctor ID %d not found; keeping previous assignment.\n", new_doc);
            }
        }
    }

    if (p->status == PATIENT_OUTPATIENT) {
        char ch = 'N';
        if (read_char_choice("Promote to Inpatient? (Y/N): ", "YN", &ch) && ch == 'Y') {
            int ward_num = 0;
            p->status = PATIENT_INPATIENT;
            if (read_int_range("Enter Ward ID to assign (0 for none): ", 0, 999999, &ward_num) && ward_num != 0) {
                for (size_t i = 0; i < wl->count; i++) {
                    if (wl->items[i].active && wl->items[i].ward_id == ward_num) {
                        if (wl->items[i].occupied_beds < wl->items[i].total_beds) {
                            wl->items[i].occupied_beds++;
                            p->assigned_ward_id = ward_num;
                        } else {
                            printf("[Warning] Ward is full.\n");
                        }
                        break;
                    }
                }
            }
        }
    }

    printf("[Success] Patient record updated.\n");
}

static void discharge_patient(PatientList *pl, WardList *wl) {
    int id = 0;
    Patient *p;

    print_header("DISCHARGE PATIENT");
    if (!read_int_range("Enter Patient ID to discharge: ", 1, 999999, &id)) {
        printf("[Error] Invalid ID.\n");
        return;
    }
    p = find_patient_by_id(pl, id);
    if (!p) {
        printf("[Error] Patient ID %d not found.\n", id);
        return;
    }
    if (p->status == PATIENT_DISCHARGED) {
        printf("[Info] Patient %s is already discharged.\n", p->full_name);
        return;
    }

    if (p->assigned_ward_id > 0) {
        for (size_t i = 0; i < wl->count; i++) {
            if (wl->items[i].active && wl->items[i].ward_id == p->assigned_ward_id) {
                if (wl->items[i].occupied_beds > 0) {
                    wl->items[i].occupied_beds--;
                }
                break;
            }
        }
        p->assigned_ward_id = -1;
    }

    p->status = PATIENT_DISCHARGED;
    printf("[Success] Patient %s (ID %d) marked as Discharged and bed released.\n", p->full_name, p->id);
}

static void sort_patients_by_name(PatientList *pl) {
    for (size_t i = 0; i < pl->count; i++) {
        for (size_t j = i + 1; j < pl->count; j++) {
            if (strcmp(pl->items[i].full_name, pl->items[j].full_name) > 0) {
                Patient tmp = pl->items[i];
                pl->items[i] = pl->items[j];
                pl->items[j] = tmp;
            }
        }
    }
}

static void sort_patients_by_id(PatientList *pl) {
    for (size_t i = 0; i < pl->count; i++) {
        for (size_t j = i + 1; j < pl->count; j++) {
            if (pl->items[i].id > pl->items[j].id) {
                Patient tmp = pl->items[i];
                pl->items[i] = pl->items[j];
                pl->items[j] = tmp;
            }
        }
    }
}

static void schedule_appointment(AppointmentList *al, PatientList *pl, DoctorList *dl) {
    Appointment a;
    memset(&a, 0, sizeof(Appointment));
    a.id = al->next_id;
    a.status = APPT_SCHEDULED;
    a.active = 1;

    print_header("SCHEDULE APPOINTMENT");
    if (!read_int_range("Patient ID: ", 1, 999999, &a.patient_id) ||
        find_patient_by_id(pl, a.patient_id) == NULL) {
        printf("[Error] Active patient not found.\n");
        return;
    }
    if (!read_int_range("Doctor ID: ", 1, 999999, &a.doctor_id) ||
        find_doctor_by_id(dl, a.doctor_id) == NULL) {
        printf("[Error] Active doctor not found.\n");
        return;
    }
    if (!read_line("Appointment Date (YYYY-MM-DD): ", a.date, sizeof(a.date)) ||
        !is_valid_date(a.date)) {
        printf("[Error] Invalid date format.\n");
        return;
    }
    if (!read_line("Time Slot (e.g., 10:30): ", a.time_slot, sizeof(a.time_slot)) ||
        a.time_slot[0] == '\0') {
        printf("[Error] Time slot cannot be empty.\n");
        return;
    }
    read_line("Reason / Visit Notes: ", a.reason, sizeof(a.reason));

    /* Check for double-booking on same doctor, date, and time */
    for (size_t i = 0; i < al->count; i++) {
        const Appointment *existing = &al->items[i];
        if (existing->active &&
            existing->status == APPT_SCHEDULED &&
            existing->doctor_id == a.doctor_id &&
            strcmp(existing->date, a.date) == 0 &&
            strcmp(existing->time_slot, a.time_slot) == 0) {
            printf("[Error] Doctor %d already has a scheduled appointment on %s at %s.\n",
                   a.doctor_id, a.date, a.time_slot);
            return;
        }
    }

    if (!ensure_appointment_capacity(al)) {
        printf("[Error] Memory allocation failed.\n");
        return;
    }
    al->items[al->count++] = a;
    al->next_id++;
    printf("[Success] Appointment booked with ID %d.\n", a.id);
}

static void assign_bed_to_patient(WardList *wl, PatientList *pl) {
    int patient_id = 0;
    int ward_num = 0;
    Patient *p;
    Ward *w;

    print_header("ASSIGN BED TO PATIENT");
    if (!read_int_range("Enter Patient ID: ", 1, 999999, &patient_id)) {
        printf("[Error] Invalid ID.\n");
        return;
    }
    p = find_patient_by_id(pl, patient_id);
    if (!p) {
        printf("[Error] Patient ID %d not found.\n", patient_id);
        return;
    }
    if (p->status == PATIENT_DISCHARGED) {
        printf("[Error] Cannot assign a bed to a discharged patient.\n");
        return;
    }
    if (p->assigned_ward_id > 0) {
        printf("[Error] Patient is already assigned to Ward %d.\n", p->assigned_ward_id);
        return;
    }
    if (!read_int_range("Enter Ward ID: ", 1, 999999, &ward_num)) {
        printf("[Error] Invalid Ward ID.\n");
        return;
    }
    w = find_ward_by_id(wl, ward_num);
    if (!w) {
        printf("[Error] Ward %d not found.\n", ward_num);
        return;
    }
    if (w->occupied_beds >= w->total_beds) {
        printf("[Error] Ward %d (%s) has no available beds.\n", w->ward_id, w->ward_name);
        return;
    }

    w->occupied_beds++;
    p->assigned_ward_id = w->ward_id;
    p->status = PATIENT_INPATIENT;
    printf("[Success] Assigned %s to Ward %d (%s).\n", p->full_name, w->ward_id, w->ward_name);
}

static void release_patient_bed(WardList *wl, PatientList *pl) {
    int patient_id = 0;
    Patient *p;
    Ward *w;

    print_header("RELEASE PATIENT BED");
    if (!read_int_range("Enter Patient ID: ", 1, 999999, &patient_id)) {
        printf("[Error] Invalid ID.\n");
        return;
    }
    p = find_patient_by_id(pl, patient_id);
    if (!p || p->assigned_ward_id <= 0) {
        printf("[Error] Patient has no active ward bed assignment.\n");
        return;
    }

    w = find_ward_by_id(wl, p->assigned_ward_id);
    if (w && w->occupied_beds > 0) {
        w->occupied_beds--;
    }
    printf("[Success] Released bed in Ward %d for %s.\n", p->assigned_ward_id, p->full_name);
    p->assigned_ward_id = -1;
    if (p->status == PATIENT_INPATIENT) {
        p->status = PATIENT_OUTPATIENT;
    }
}

static void create_bill(BillList *bl, PatientList *pl, DoctorList *dl, WardList *wl) {
    Bill b;
    Patient *p;
    Doctor *d = NULL;
    Ward *w = NULL;
    int days_stayed = 0;
    double subtotal = 0.0;

    memset(&b, 0, sizeof(Bill));
    b.id = bl->next_id;
    b.active = 1;

    print_header("GENERATE PATIENT BILL");
    if (!read_int_range("Enter Patient ID: ", 1, 999999, &b.patient_id)) {
        printf("[Error] Invalid ID.\n");
        return;
    }
    p = find_patient_by_id(pl, b.patient_id);
    if (!p) {
        printf("[Error] Patient ID %d not found.\n", b.patient_id);
        return;
    }

    get_current_date(b.billing_date, sizeof(b.billing_date));

    if (p->assigned_doctor_id > 0) {
        d = find_doctor_by_id(dl, p->assigned_doctor_id);
    }
    if (d) {
        printf("Assigned Doctor: %s (Default Fee: $%.2f)\n", d->full_name, d->consultation_fee);
        b.consultation_charge = d->consultation_fee;
    }
    if (!read_double_min("Consultation Charge ($): ", 0.0, &b.consultation_charge)) {
        printf("[Error] Invalid consultation charge.\n");
        return;
    }

    if (p->assigned_ward_id > 0) {
        w = find_ward_by_id(wl, p->assigned_ward_id);
    }
    if (w && read_int_range("Number of Ward Days to Bill: ", 0, 365, &days_stayed)) {
        b.room_charge = days_stayed * w->daily_rate;
        printf("Calculated Ward Charge (%d days @ $%.2f): $%.2f\n",
               days_stayed, w->daily_rate, b.room_charge);
    } else if (!read_double_min("Room / Ward Charge ($): ", 0.0, &b.room_charge)) {
        printf("[Error] Invalid room charge.\n");
        return;
    }

    if (!read_double_min("Medication Charge ($): ", 0.0, &b.medication_charge) ||
        !read_double_min("Laboratory / Diagnostics Charge ($): ", 0.0, &b.lab_charge) ||
        !read_double_min("Discount / Insurance Waiver ($): ", 0.0, &b.discount_amount)) {
        printf("[Error] Invalid charge amount.\n");
        return;
    }

    subtotal = b.consultation_charge + b.room_charge + b.medication_charge + b.lab_charge;
    if (b.discount_amount > subtotal) {
        b.discount_amount = subtotal;
    }
    b.total_amount = subtotal - b.discount_amount;

    if (!read_double_min("Initial Payment Received Today ($): ", 0.0, &b.amount_paid)) {
        printf("[Error] Invalid payment amount.\n");
        return;
    }
    if (b.amount_paid > b.total_amount) {
        b.amount_paid = b.total_amount;
    }
    b.is_settled = (b.amount_paid >= b.total_amount - 0.005) ? 1 : 0;

    if (!ensure_bill_capacity(bl)) {
        printf("[Error] Memory allocation failed.\n");
        return;
    }
    bl->items[bl->count++] = b;
    bl->next_id++;

    printf("\nBill #%d created for %s (Total: $%.2f | Paid: $%.2f | Balance: $%.2f)\n",
           b.id, p->full_name, b.total_amount, b.amount_paid, b.total_amount - b.amount_paid);
}

static void list_bills(const BillList *bl, const PatientList *pl) {
    size_t count = 0;
    double grand_total = 0.0;
    double grand_paid = 0.0;

    print_header("ALL PATIENT BILLS");
    for (size_t i = 0; i < bl->count; i++) {
        const Bill *b = &bl->items[i];
        const Patient *p;
        double balance;
        if (!b->active) continue;
        p = find_patient_by_id((PatientList *)pl, b->patient_id);
        balance = b->total_amount - b->amount_paid;
        grand_total += b->total_amount;
        grand_paid += b->amount_paid;

        printf("Bill #%-5d | Date: %s | Patient: %-20s (ID %d)\n",
               b->id, b->billing_date, p ? p->full_name : "Unknown", b->patient_id);
        printf("   Consult: $%.2f | Room: $%.2f | Meds: $%.2f | Lab: $%.2f | Disc: -$%.2f\n",
               b->consultation_charge, b->room_charge, b->medication_charge, b->lab_charge, b->discount_amount);
        printf("   Total: $%.2f | Paid: $%.2f | Balance: $%.2f | Status: %s\n",
               b->total_amount, b->amount_paid, balance > 0.0 ? balance : 0.0,
               b->is_settled ? "SETTLED" : "DUE");
        print_separator('-', 52);
        count++;
    }
    if (count == 0) {
        printf("No bills generated yet.\n");
    } else {
        printf("Bills: %zu | Total Billed: $%.2f | Collected: $%.2f | Outstanding: $%.2f\n",
               count, grand_total, grand_paid, grand_total - grand_paid);
    }
}

static void record_bill_payment(BillList *bl, const PatientList *pl) {
    int bill_id = 0;
    double payment = 0.0;
    double balance = 0.0;
    Bill *b;
    const Patient *p;

    print_header("RECORD BILL PAYMENT");
    if (!read_int_range("Enter Bill ID: ", 1, 999999, &bill_id)) {
        printf("[Error] Invalid ID.\n");
        return;
    }
    b = find_bill_by_id(bl, bill_id);
    if (!b) {
        printf("[Error] Bill #%d not found.\n", bill_id);
        return;
    }
    balance = b->total_amount - b->amount_paid;
    if (b->is_settled || balance <= 0.005) {
        printf("[Info] Bill #%d is already settled.\n", b->id);
        return;
    }

    p = find_patient_by_id((PatientList *)pl, b->patient_id);
    printf("Patient: %s | Total: $%.2f | Paid: $%.2f | Remaining Balance: $%.2f\n",
           p ? p->full_name : "Unknown", b->total_amount, b->amount_paid, balance);

    if (!read_double_min("Payment Amount ($): ", 0.01, &payment)) {
        printf("[Error] Invalid payment amount.\n");
        return;
    }
    if (payment > balance) {
        printf("[Warning] Payment exceeds balance; capping at $%.2f.\n", balance);
        payment = balance;
    }

    b->amount_paid += payment;
    if (b->amount_paid >= b->total_amount - 0.005) {
        b->amount_paid = b->total_amount;
        b->is_settled = 1;
    }
    printf("[Success] Payment recorded. New balance: $%.2f (%s)\n",
           b->total_amount - b->amount_paid, b->is_settled ? "SETTLED" : "DUE");
}

static void view_unsettled_bills(const BillList *bl, const PatientList *pl) {
    size_t count = 0;
    double total_due = 0.0;

    print_header("UNSETTLED BILLS");
    for (size_t i = 0; i < bl->count; i++) {
        const Bill *b = &bl->items[i];
        const Patient *p;
        double balance;
        if (!b->active || b->is_settled) continue;
        balance = b->total_amount - b->amount_paid;
        if (balance <= 0.005) continue;
        p = find_patient_by_id((PatientList *)pl, b->patient_id);
        printf("  Bill #%-5d | %-20s (ID %d) | Balance Due: $%.2f\n",
               b->id, p ? p->full_name : "Unknown", b->patient_id, balance);
        total_due += balance;
        count++;
    }
    printf("Unsettled Bills: %zu | Total Amount Due: $%.2f\n", count, total_due);
}

static void reports_menu(const PatientList *pl, const DoctorList *dl, const AppointmentList *al, const WardList *wl, const BillList *bl) {
    int choice = 0;
    do {
        print_header("REPORTS & ANALYTICS");
        printf("  1. View Hospital Summary Dashboard\n");
        printf("  2. Export Summary Report to Text File (%s)\n", FILE_REPORT_TXT);
        printf("  0. Back to Main Menu\n");
        print_separator('-', 52);

        if (!read_int_range("Select option: ", 0, 2, &choice)) {
            printf("[Error] Enter a valid option (0-2).\n");
            continue;
        }

        switch (choice) {
            case 1:
                print_hospital_dashboard(pl, dl, al, wl, bl);
                pause_for_enter();
                break;
            case 2:
                export_summary_report(pl, dl, al, wl, bl, FILE_REPORT_TXT);
                pause_for_enter();
                break;
            case 0:
                break;
            default:
                break;
        }
    } while (choice != 0);
}

static void print_hospital_dashboard(const PatientList *pl, const DoctorList *dl, const AppointmentList *al, const WardList *wl, const BillList *bl) {
    int outpatients = 0, inpatients = 0, discharged = 0;
    int avail_docs = 0, total_docs = 0;
    int sched_appts = 0, comp_appts = 0, canc_appts = 0;
    int total_beds = 0, occ_beds = 0;
    double billed = 0.0, collected = 0.0;

    for (size_t i = 0; i < pl->count; i++) {
        if (!pl->items[i].active) continue;
        if (pl->items[i].status == PATIENT_OUTPATIENT) outpatients++;
        else if (pl->items[i].status == PATIENT_INPATIENT) inpatients++;
        else if (pl->items[i].status == PATIENT_DISCHARGED) discharged++;
    }

    for (size_t i = 0; i < dl->count; i++) {
        if (!dl->items[i].active) continue;
        total_docs++;
        if (dl->items[i].available) avail_docs++;
    }

    for (size_t i = 0; i < al->count; i++) {
        if (!al->items[i].active) continue;
        if (al->items[i].status == APPT_SCHEDULED) sched_appts++;
        else if (al->items[i].status == APPT_COMPLETED) comp_appts++;
        else if (al->items[i].status == APPT_CANCELLED) canc_appts++;
    }

    for (size_t i = 0; i < wl->count; i++) {
        if (!wl->items[i].active) continue;
        total_beds += wl->items[i].total_beds;
        occ_beds += wl->items[i].occupied_beds;
    }

    for (size_t i = 0; i < bl->count; i++) {
        if (!bl->items[i].active) continue;
        billed += bl->items[i].total_amount;
        collected += bl->items[i].amount_paid;
    }

    print_header("HOSPITAL SUMMARY DASHBOARD");
    printf("Patients      : %d Total (%d Inpatient, %d Outpatient, %d Discharged)\n",
           outpatients + inpatients + discharged, inpatients, outpatients, discharged);
    printf("Doctors       : %d Total (%d Available)\n", total_docs, avail_docs);
    printf("Appointments  : %d Scheduled, %d Completed, %d Cancelled\n",
           sched_appts, comp_appts, canc_appts);
    printf("Bed Occupancy : %d / %d beds occupied (%.1f%%)\n",
           occ_beds, total_beds,
           (total_beds > 0) ? (100.0 * (double)occ_beds / (double)total_beds) : 0.0);
    printf("Financials    : Billed $%.2f | Collected $%.2f | Outstanding $%.2f\n",
           billed, collected, billed - collected);
}

static void export_summary_report(const PatientList *pl, const DoctorList *dl, const AppointmentList *al, const WardList *wl, const BillList *bl, const char *filename) {
    FILE *fp = fopen(filename, "w");
    char today[MAX_DATE_LEN];
    double billed = 0.0;
    double collected = 0.0;

    if (!fp) {
        printf("[Error] Could not create report file '%s'.\n", filename);
        return;
    }

    get_current_date(today, sizeof(today));
    fprintf(fp, "========================================================\n");
    fprintf(fp, "HOSPITAL MANAGEMENT SYSTEM - SUMMARY REPORT (%s)\n", today);
    fprintf(fp, "========================================================\n\n");

    fprintf(fp, "--- PATIENTS (%zu records) ---\n", pl->count);
    for (size_t i = 0; i < pl->count; i++) {
        if (pl->items[i].active) {
            fprintf(fp, "ID: %d | Name: %s | Status: %s | Dx: %s\n",
                    pl->items[i].id, pl->items[i].full_name,
                    patient_status_str(pl->items[i].status), pl->items[i].diagnosis);
        }
    }

    fprintf(fp, "\n--- DOCTORS (%zu records) ---\n", dl->count);
    for (size_t i = 0; i < dl->count; i++) {
        if (dl->items[i].active) {
            fprintf(fp, "ID: %d | Name: %s | Spec: %s | Fee: $%.2f | %s\n",
                    dl->items[i].id, dl->items[i].full_name, dl->items[i].specialization,
                    dl->items[i].consultation_fee, dl->items[i].available ? "Available" : "Unavailable");
        }
    }

    fprintf(fp, "\n--- WARDS (%zu records) ---\n", wl->count);
    for (size_t i = 0; i < wl->count; i++) {
        if (wl->items[i].active) {
            fprintf(fp, "Ward %d (%s): %d/%d occupied | $%.2f/day\n",
                    wl->items[i].ward_id, wl->items[i].ward_name,
                    wl->items[i].occupied_beds, wl->items[i].total_beds, wl->items[i].daily_rate);
        }
    }

    fprintf(fp, "\n--- APPOINTMENTS (%zu records) ---\n", al->count);
    for (size_t i = 0; i < al->count; i++) {
        if (al->items[i].active) {
            fprintf(fp, "Appt #%d | %s %s | Patient ID: %d | Doctor ID: %d | %s\n",
                    al->items[i].id, al->items[i].date, al->items[i].time_slot,
                    al->items[i].patient_id, al->items[i].doctor_id,
                    appointment_status_str(al->items[i].status));
        }
    }

    fprintf(fp, "\n--- BILLING (%zu records) ---\n", bl->count);
    for (size_t i = 0; i < bl->count; i++) {
        if (bl->items[i].active) {
            billed += bl->items[i].total_amount;
            collected += bl->items[i].amount_paid;
            fprintf(fp, "Bill #%d | Patient ID: %d | Total: $%.2f | Paid: $%.2f | %s\n",
                    bl->items[i].id, bl->items[i].patient_id,
                    bl->items[i].total_amount, bl->items[i].amount_paid,
                    bl->items[i].is_settled ? "SETTLED" : "DUE");
        }
    }
    fprintf(fp, "\nTotal Billed: $%.2f | Total Collected: $%.2f | Outstanding: $%.2f\n",
            billed, collected, billed - collected);

    fclose(fp);
    printf("[Success] Summary report exported to '%s'.\n", filename);
}
